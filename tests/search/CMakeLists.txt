cmake_minimum_required(VERSION 3.13)
project(search_tests LANGUAGES CXX)
enable_testing()
find_package(ICU REQUIRED COMPONENTS uc)
find_package(pugixml REQUIRED)
option(SEARCH_SANITIZERS "Run memory/undefined behaviour checks" OFF)
add_library(search_test_core STATIC ../../es-app/src/search/SearchCore.cpp)
target_include_directories(search_test_core PUBLIC ../../es-app/src)
target_compile_features(search_test_core PUBLIC cxx_std_17)
target_compile_definitions(search_test_core PRIVATE SEARCH_TEST_BUILD_TIMING)
target_compile_options(search_test_core PRIVATE -Wall -Wextra -Werror)
target_link_libraries(search_test_core PUBLIC ICU::uc)
if(SEARCH_SANITIZERS)
    target_compile_options(search_test_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(search_test_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(search_tests main.cpp
               ../../es-app/src/search/SearchHistory.cpp)
target_include_directories(search_tests PRIVATE ../../es-app/src)
target_compile_features(search_tests PRIVATE cxx_std_17)
target_compile_definitions(search_tests PRIVATE SEARCH_REAL_TITLES="${CMAKE_CURRENT_SOURCE_DIR}/real-titles.txt")
target_compile_options(search_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(search_tests PRIVATE search_test_core ICU::uc pugixml::pugixml)
if(SEARCH_SANITIZERS)
    target_compile_options(search_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(search_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME search_logic COMMAND search_tests)
add_test(NAME search_performance COMMAND search_tests --performance)
add_executable(search_adapter_tests adapter.cpp ../../es-app/src/search/LibrarySearch.cpp
    ../../es-app/src/search/SearchHistory.cpp
    ../../es-app/src/MetaData.cpp)
target_include_directories(search_adapter_tests PRIVATE frontend ../../es-app/src)
target_compile_features(search_adapter_tests PRIVATE cxx_std_17)
target_compile_options(search_adapter_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(search_adapter_tests PRIVATE search_test_core ICU::uc pugixml::pugixml)
if(SEARCH_SANITIZERS)
    target_compile_options(search_adapter_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(search_adapter_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME search_adapter COMMAND search_adapter_tests)

set_tests_properties(search_logic PROPERTIES TIMEOUT 30)
# The adapter now includes a 50k snapshot; Debug sanitizer instrumentation has no frame-time gate.
set_tests_properties(search_adapter PROPERTIES TIMEOUT 180)
set_tests_properties(search_performance PROPERTIES TIMEOUT 60)

add_executable(search_realistic realistic.cpp)
target_include_directories(search_realistic PRIVATE ../../es-app/src)
target_compile_features(search_realistic PRIVATE cxx_std_17)
target_compile_options(search_realistic PRIVATE -Wall -Wextra -Werror)
target_link_libraries(search_realistic PRIVATE search_test_core ICU::uc pugixml::pugixml)
target_compile_definitions(search_realistic PRIVATE SEARCH_REAL_TITLES="${CMAKE_CURRENT_SOURCE_DIR}/real-titles.txt" SEARCH_MAME_TITLES="${CMAKE_CURRENT_SOURCE_DIR}/../../resources/MAME/mamenames.xml")
add_test(NAME search_realistic_performance COMMAND search_realistic)
add_test(NAME search_memory COMMAND search_realistic --memory)
add_test(NAME search_screen_realistic COMMAND search_realistic --screen)
set_tests_properties(search_realistic_performance search_screen_realistic search_memory PROPERTIES TIMEOUT 120)
add_executable(search_fuzz fuzz.cpp)
target_include_directories(search_fuzz PRIVATE ../../es-app/src)
target_compile_features(search_fuzz PRIVATE cxx_std_17)
target_compile_options(search_fuzz PRIVATE -Wall -Wextra -Werror)
target_link_libraries(search_fuzz PRIVATE search_test_core ICU::uc)
if(SEARCH_SANITIZERS)
    target_compile_options(search_fuzz PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(search_fuzz PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME search_fuzz COMMAND search_fuzz)
set_tests_properties(search_fuzz PROPERTIES TIMEOUT 120)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/search_hooks.inc"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_hooks.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/../.." "${CMAKE_CURRENT_BINARY_DIR}/search_hooks.inc"
    DEPENDS extract_hooks.py ../../es-app/src/FileData.cpp ../../es-app/src/SystemData.cpp)
add_executable(search_hook_tests hooks.cpp "${CMAKE_CURRENT_BINARY_DIR}/search_hooks.inc")
target_include_directories(search_hook_tests PRIVATE ../../es-app/src "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(search_hook_tests PRIVATE cxx_std_17)
target_compile_options(search_hook_tests PRIVATE -Wall -Wextra -Werror)
if(SEARCH_SANITIZERS)
    target_compile_options(search_hook_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(search_hook_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME search_hooks COMMAND search_hook_tests)

set_tests_properties(search_performance search_realistic_performance PROPERTIES ENVIRONMENT "SEARCH_BUILD_PROFILE=1")
